🧪 Tests Unitarios con Jest y POO

🎯 ¿Qué son los Tests Unitarios?

Los tests unitarios verifican que cada parte individual (unidad) de tu código funcione correctamente. Con POO, testeamos clases, métodos y sus interacciones.

1

📦 Instalación y Configuración de Jest

                
                    

                        // En tu terminal
                        npm init -y
                        npm install --save-dev jest

                        // package.json - agregar scripts
                        {
                            "scripts": {
                                "test": "jest",
                                "test:watch": "jest --watch"
                            }
                        }

                        // jest.config.js (opcional)
                        module.exports = {
                            testEnvironment: 'node',
                            collectCoverage: true,
                            coverageDirectory: 'coverage'
                        };

🏗️ Crear Clases para Testear (POO)

                
                    

                        // calculadora.js
                        class Calculadora {
                            constructor() {
                                this.historial = [];
                            }

                            sumar(a, b) {
                                const resultado = a + b;
                                this.historial.push(`Suma: ${a} + ${b} = ${resultado}`);
                                return resultado;
                            }

                            restar(a, b) {
                                const resultado = a - b;
                                this.historial.push(`Resta: ${a} - ${b} = ${resultado}`);
                                return resultado;
                            }

                            multiplicar(a, b) {
                                if (b === 0) {
                                    throw new Error('No se puede multiplicar por cero');
                                }
                                const resultado = a * b;
                                this.historial.push(`Multiplicación: ${a} * ${b} = ${resultado}`);
                                return resultado;
                            }

                            obtenerHistorial() {
                                return this.historial;
                            }

                            limpiarHistorial() {
                                this.historial = [];
                            }
                        }

                        module.exports = Calculadora;
3

🧪 Crear Tests Básicos para la Clase

                
                    

                        // calculadora.test.js
                        const Calculadora = require('./calculadora');

                        describe('Calculadora', () => {
                            let calculadora;

                            // beforeEach se ejecuta antes de cada test
                            beforeEach(() => {
                                calculadora = new Calculadora();
                            });

                            test('debería sumar correctamente', () => {
                                expect(calculadora.sumar(2, 3)).toBe(5);
                                expect(calculadora.sumar(-1, 1)).toBe(0);
                            });

                            test('debería restar correctamente', () => {
                                expect(calculadora.restar(5, 3)).toBe(2);
                                expect(calculadora.restar(3, 5)).toBe(-2);
                            });

                            test('debería lanzar error al multiplicar por cero', () => {
                                expect(() => calculadora.multiplicar(5, 0)).toThrow('No se puede multiplicar por cero');
                            });
                        });

📊 Testear Estado y Comportamiento de Objetos

                
                    

                        describe('Gestión del Historial', () => {
                            let calculadora;

                            beforeEach(() => {
                                calculadora = new Calculadora();
                            });

                            test('debería guardar operaciones en el historial', () => {
                                calculadora.sumar(2, 3);
                                calculadora.restar(5, 2);
                                
                                const historial = calculadora.obtenerHistorial();
                                
                                expect(historial).toHaveLength(2);
                                expect(historial[0]).toContain('Suma: 2 + 3 = 5');
                                expect(historial[1]).toContain('Resta: 5 - 2 = 3');
                            });

                            test('debería limpiar el historial correctamente', () => {
                                calculadora.sumar(1, 1);
                                expect(calculadora.obtenerHistorial()).toHaveLength(1);
                                
                                calculadora.limpiarHistorial();
                                expect(calculadora.obtenerHistorial()).toHaveLength(0);
                            });
                        });

🔗 Testear Herencia y Polimorfismo

                
                    

                        // calculadora-cientifica.js
                        const Calculadora = require('./calculadora');

                        class CalculadoraCientifica extends Calculadora {
                            potencia(base, exponente) {
                                const resultado = Math.pow(base, exponente);
                                this.historial.push(`Potencia: ${base}^${exponente} = ${resultado}`);
                                return resultado;
                            }

                            raizCuadrada(numero) {
                                if (numero < 0) {
                                    throw new Error('No se puede calcular raíz de número negativo');
                                }
                                const resultado = Math.sqrt(numero);
                                this.historial.push(`Raíz cuadrada: √${numero} = ${resultado}`);
                                return resultado;
                            }
                        }

                        module.exports = CalculadoraCientifica;
                    

                
            
                
                    

                        // calculadora-cientifica.test.js
                        const CalculadoraCientifica = require('./calculadora-cientifica');

                        describe('CalculadoraCientifica', () => {
                            let calcCientifica;

                            beforeEach(() => {
                                calcCientifica = new CalculadoraCientifica();
                            });

                            test('debería heredar métodos de Calculadora', () => {
                                expect(calcCientifica.sumar(2, 2)).toBe(4);
                                expect(calcCientifica.restar(5, 3)).toBe(2);
                            });

                            test('debería calcular potencias correctamente', () => {
                                expect(calcCientifica.potencia(2, 3)).toBe(8);
                                expect(calcCientifica.potencia(5, 2)).toBe(25);
                            });

                            test('debería lanzar error con raíz de número negativo', () => {
                                expect(() => calcCientifica.raizCuadrada(-4)).toThrow();
                            });
                        });

🎭 Mocks y Dependencias

                
                    

                        // servicio-db.js
                        class ServicioDB {
                            guardar(operacion, resultado) {
                                // Simula guardar en base de datos
                                console.log(`Guardando: ${operacion} = ${resultado}`);
                                return true;
                            }

                            obtenerHistorial() {
                                // Simula obtener de base de datos
                                return ['2+2=4', '3*3=9'];
                            }
                        }

                        // calculadora-con-db.js
                        class CalculadoraConDB {
                            constructor(servicioDB) {
                                this.servicioDB = servicioDB;
                            }

                            sumar(a, b) {
                                const resultado = a + b;
                                this.servicioDB.guardar(`${a}+${b}`, resultado);
                                return resultado;
                            }
                        }
                    

                
            
                
                    

                        // calculadora-con-db.test.js
                        const CalculadoraConDB = require('./calculadora-con-db');

                        describe('CalculadoraConDB con Mocks', () => {
                            test('debería usar el servicio DB al sumar', () => {
                                // Crear mock del servicio
                                const mockServicioDB = {
                                    guardar: jest.fn() // función mockeada
                                };

                                const calculadora = new CalculadoraConDB(mockServicioDB);
                                const resultado = calculadora.sumar(2, 3);

                                expect(resultado).toBe(5);
                                expect(mockServicioDB.guardar).toHaveBeenCalledWith('2+3', 5);
                                expect(mockServicioDB.guardar).toHaveBeenCalledTimes(1);
                            });
                        });

🚀 Comandos para Ejecutar Tests

            
                

                    # Ejecutar todos los tests
                    npm test

                    # Ejecutar en modo watch (se actualiza automáticamente)
                    npm run test:watch

                    # Ejecutar tests específicos
                    npm test -- calculadora.test.js

                    # Ver cobertura de código
                    npm test -- --coverage

💡 Buenas Prácticas para Tests con POO

  • Un test por comportamiento - No mezcles múltiples pruebas en un solo test
  • Usa beforeEach/afterEach - Para configurar y limpiar entre tests
  • Testea métodos públicos - No testees métodos privados/internalos
  • Nombres descriptivos - describe('Calculadora') y test('debería sumar correctamente')
  • Mocks para dependencias externas - Base de datos, APIs, etc.
